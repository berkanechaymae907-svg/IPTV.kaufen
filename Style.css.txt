    <script>
        // Fonction pour créer un bouton PayPal
        function createPayPalButton(containerId, amount) {
            paypal.Buttons({
                createOrder: function(data, actions) {
                    return actions.order.create({
                        purchase_units: [{
                            amount: {
                                value: amount,
                                currency_code: 'EUR'
                            }
                        }]
                    });
                },
                onApprove: function(data, actions) {
                    return actions.order.capture().then(function(details) {
                        alert('Transaktion abgeschlossen von ' + details.payer.name.given_name);
                        // Optionnel: Rediriger vers WhatsApp pour la livraison
                        window.open('https://wa.me/DEINENUMMER?text=Ich habe das Abonnement für ' + amount + '€ bezahlt.', '_blank');
                    });
                }
            }).render('#' + containerId);
        }

        // Initialiser les boutons pour chaque forfait
        createPayPalButton('paypal-button-container-6m', '30.00');
        createPayPalButton('paypal-button-container-1y', '40.00');
        createPayPalButton('paypal-button-container-2y', '70.00');
